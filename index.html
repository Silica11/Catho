function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetUS = ss.getSheetByName("Dati_US"); var sheetStrut = ss.getSheetByName("Struttura"); var sheetUtenti = ss.getSheetByName("utenti"); var dataUS = sheetUS ? getSheetData(sheetUS) : []; var dataStrut = sheetStrut ? getSheetData(sheetStrut) : []; var strutturaObj = {}; dataStrut.forEach(function(row) { var s = row.Scavo || row.scavo || row["Scavo"]; var c = row.Cantiere || row.cantiere || row["Cantiere"]; var q = row.Quadrante || row.quadrante || row["Quadrante"]; if (!s) return; if (!strutturaObj[s]) strutturaObj[s] = {}; if (c && !strutturaObj[s][c]) strutturaObj[s][c] = []; if (q && c && strutturaObj[s][c].indexOf(q) === -1) strutturaObj[s][c].push(q); }); var utentiList = []; if (sheetUtenti) { var uData = sheetUtenti.getDataRange().getValues(); for (var i = 1; i < uData.length; i++) { if (uData[i][0]) utentiList.push(uData[i][0]); } } return ContentService.createTextOutput(JSON.stringify({ us: dataUS, strutturaObj: strutturaObj, utenti: utentiList })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.action === "REGISTER_USER") { var sheetUtenti = ss.getSheetByName("utenti"); if (!sheetUtenti) { sheetUtenti = ss.insertSheet("utenti"); sheetUtenti.appendRow(["Nome", "Data Registrazione"]); } sheetUtenti.appendRow([data.nome, new Date()]); return ContentService.createTextOutput(JSON.stringify({"result":"success"})).setMimeType(ContentService.MimeType.JSON); } if (data.action === "SAVE_STRUTTURA") { var sheetStrut = ss.getSheetByName("Struttura"); if (!sheetStrut) sheetStrut = ss.insertSheet("Struttura"); sheetStrut.clearContents(); sheetStrut.appendRow(["Scavo", "Cantiere", "Quadrante"]); for (var s in data.struttura) { var cantieri = data.struttura[s]; var haCantieri = false; for (var c in cantieri) { haCantieri = true; var quadranti = cantieri[c]; if (quadranti.length === 0) { sheetStrut.appendRow([s, c, ""]); } else { quadranti.forEach(function(q) { sheetStrut.appendRow([s, c, q]); }); } } if (!haCantieri) sheetStrut.appendRow([s, "", ""]); } return ContentService.createTextOutput(JSON.stringify({"result":"success"})).setMimeType(ContentService.MimeType.JSON); } if (data.action === "DELETE_US") { var sheet = ss.getSheetByName("Dati_US"); var values = sheet.getDataRange().getValues(); for (var i = values.length - 1; i >= 1; i--) { if (values[i][1] == data.scavo && values[i][2] == data.cantiere && values[i][3] == data.quadrante && values[i][4] == data.numero_us) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({"result":"success"})).setMimeType(ContentService.MimeType.JSON); } var folder = DriveApp.getFolderById("10MYSnmg-5YRaU6LdLNXo6D9XvV6rtNhp"); var sheet = ss.getSheetByName("Dati_US"); if(!sheet) sheet = ss.insertSheet("Dati_US"); var values = sheet.getDataRange().getValues(); var rigaTrovata = -1; for (var i = 1; i < values.length; i++) { if (values[i][1] == data.scavo && values[i][2] == data.cantiere && values[i][3] == data.quadrante && values[i][4] == data.numero_us) { rigaTrovata = i + 1; break; } } // GESTIONE UNIVERSALE FILE (Mantiene l'estensione corretta) var fileUrl = ""; if (data.fotoFile && data.fotoFile.base64) { var estensione = ".jpg"; if(data.fotoFile.name && data.fotoFile.name.includes('.')) { estensione = data.fotoFile.name.substring(data.fotoFile.name.lastIndexOf('.')); } var blob = Utilities.newBlob(Utilities.base64Decode(data.fotoFile.base64), data.fotoFile.type, "US_" + data.numero_us + estensione); var file = folder.createFile(blob); fileUrl = file.getUrl(); } else if (rigaTrovata != -1 && values[rigaTrovata - 1]) { fileUrl = values[rigaTrovata - 1][8] || ""; } var rapp = { "Copre": "", "È coperto da": "", "Taglia": "", "È tagliato da": "", "Riempie": "", "È riempito da": "" }; if (data.rapporti) { data.rapporti.forEach(function(r) { if (rapp[r.tipo] !== undefined) rapp[r.tipo] += (rapp[r.tipo] === "" ? "" : ", ") + r.num; }); } // RIGA CON AGGIUNTA QUOTE (Colonne 18 e 19) var rigaDati = [ new Date(), data.scavo, data.cantiere, data.quadrante, data.numero_us, data.tipo_us, data.operatore, data.fotoSpunta ? "SÌ" : "NO", fileUrl, rapp["Copre"], rapp["È coperto da"], rapp["Taglia"], rapp["È tagliato da"], rapp["Riempie"], rapp["È riempito da"], data.descrizione, data.interpretazione, data.quota_sup, data.quota_inf ]; if (rigaTrovata != -1) { sheet.getRange(rigaTrovata, 1, 1, rigaDati.length).setValues([rigaDati]); } else { sheet.appendRow(rigaDati); } return ContentService.createTextOutput(JSON.stringify({"result":"success"})).setMimeType(ContentService.MimeType.JSON); } catch (f) { return ContentService.createTextOutput(JSON.stringify({"result":"error", "error": f.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function getSheetData(sheet) { var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var headers = values[0]; var data = []; for (var i = 1; i < values.length; i++) { var record = {}; for (var j = 0; j < headers.length; j++) record[headers[j]] = values[i][j]; data.push(record); } return data; }
